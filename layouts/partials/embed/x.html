{{- /* X（Twitter）帖子卡片，由 _markup/render-codeblock-x.html 调用。 参数：dict "url" 帖子链接或纯数字 ID，"pos" 代码块在 Markdown 里的位置（报警用）。 数据走 FxTwitter 的公开 API（api.fxtwitter.com/status/），构建时取一次。 没用 X 官方的 syndication 接口：那个接口长帖只给前 280 字，卡片上就只能 挂一个「Show more」，而博客引用一条帖子，要的恰恰是全文。 拿不到数据（帖子删了、设了私密、API 挂了）不让构建失败，退成一张只有 链接的卡片，同时 warnf 一句，构建日志里能看到是哪条。 */ -}} {{- $src := .url -}} {{- $handle := "" -}} {{- $id := "" -}} {{- with findRESubmatch `^(?:https?://)?(?:[\w-]+\.)*(?:x|twitter|fxtwitter|vxtwitter|fixupx)\.com/(\w+)/status(?:es)?/(\d+)` $src 1 -}} {{- $handle = index . 0 1 -}} {{- $id = index . 0 2 -}} {{- else -}} {{- with findRESubmatch `^(\d+)$` $src 1 -}}{{- $id = index . 0 1 -}}{{- end -}} {{- end -}} {{- $t := dict -}} {{- if $id -}} {{- with try (resources.GetRemote (printf "https://api.fxtwitter.com/status/%s" $id)) -}} {{- with .Err -}} {{- warnf "[x] %s: 请求 FxTwitter 失败：%s" $.pos . -}} {{- else with .Value -}} {{- /* 帖子不存在时 FxTwitter 回的是 200 + 一页 HTML，不是 404，所以要看类型 */ -}} {{- if eq .MediaType.SubType "json" -}} {{- $d := transform.Unmarshal . -}} {{- if eq (int $d.code) 200 -}}{{- $t = $d.tweet -}}{{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- if not $t -}}{{- warnf "[x] %s: 拿不到帖子 %s，降级成链接卡片" .pos $src -}}{{- end -}} {{- else -}} {{- warnf "[x] %s: 认不出的帖子链接 %q" .pos $src -}} {{- end -}} {{- with $t -}} {{- $url := .url -}} {{- $author := .author -}} {{- /* 静态站拿不到读者的时区，按东八区出日期（和文章日期一致，只显示到天） */ -}} {{- $date := (time.AsTime (int .created_timestamp)).UTC -}} {{- else -}} {{- /* 降级卡片。纯 ID 没有用户名，用 /i/status/，X 会自己跳到正确的地址 */ -}} {{- $href := "" -}} {{- if $id -}}{{- $href = printf "https://x.com/%s/status/%s" ($handle | default "i") $id -}} {{- else if hasPrefix $src "http" -}}{{- $href = $src -}}{{- end -}} {{- end -}}
