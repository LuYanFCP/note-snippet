{{- /* Open Graph + Twitter/X 卡片。替代 Hugo 内置的 opengraph / twitter_cards 模板：内置那套没有 og:image 时只给 summary 小卡（X 上几乎看不出来）， description 又直接拿 .Summary，会把 alert 标题「Note | 前言」一起带进去。 */ -}} {{- $title := cond .IsHome site.Title (printf "%s - %s" .Title site.Title) -}} {{- if .IsPage -}}{{- $title = .Title -}}{{- end -}} {{- $desc := .Params.description -}} {{- if and (not $desc) .IsPage -}} {{- /* 去掉 alert 的标题栏（render-blockquote 输出的「Note | 前言」） */ -}} {{- $desc = .Summary | plainify | replaceRE `(?i)\b(note|tip|important|warning|caution|insight)\s*\|\s*\S+` "" -}} {{- $desc = $desc | replaceRE `\s+` " " | strings.TrimSpace | truncate 140 -}} {{- end -}} {{- $desc = $desc | default site.Params.description -}} {{- $img := partial "og-image.html" . -}} {{- if .IsPage -}} {{- range .Params.tags -}} {{- end -}} {{- end -}} {{- with site.Params.twitter -}} {{- end -}} {{- with $img -}} {{- end -}}
